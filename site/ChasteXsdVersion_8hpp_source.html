38#define CHASTEXSDVERSION_HPP_
39
40#include <xsd/cxx/version.hxx>
41
42#include "Exception.hpp"
43
44/*
45 * Compatibility layer for CodeSynthesis XSD version macros.
46 *
47 * XSD 4.2+ provides:
48 *
49 * LIBXSD_VERSION_MAJOR
50 * LIBXSD_VERSION_MINOR
51 * LIBXSD_VERSION_PATCH
52 *
53 * Older XSD versions provide:
54 *
55 * XSD_INT_VERSION
56 *
57 * where XSD_INT_VERSION has format AABBCCDD:
58 *
59 * AA - major version
60 * BB - minor version
61 * CC - patch version
62 * DD - alpha/beta version, or 00 for a final release
63 */
64
65#if defined(LIBXSD_VERSION_MAJOR) && \
66 defined(LIBXSD_VERSION_MINOR) && \
67 defined(LIBXSD_VERSION_PATCH)
68
69 #define CHASTE_XSD_VERSION_MAJOR LIBXSD_VERSION_MAJOR
70 #define CHASTE_XSD_VERSION_MINOR LIBXSD_VERSION_MINOR
71 #define CHASTE_XSD_VERSION_PATCH LIBXSD_VERSION_PATCH
72
73#elif defined(XSD_INT_VERSION)
74
75 /*
76 * Pre-release old-style versions have DD != 00. This is currently unsupported here because:
77 * 1. it's very unlikely indeed that someone is using a pre-release XSD < 4.2
78 * 2. the old AABBCCDD encoding decrements AABBCC for alpha/beta versions, making
79 * major/minor/patch extraction very fiddly.
80 */
81 #if (XSD_INT_VERSION % 100L) != 0L
82 EXCEPTION("Unsupported pre-release CodeSynthesis XSD version");
83 #endif
84
85 #define CHASTE_XSD_VERSION_MAJOR \
86 ((XSD_INT_VERSION / 1000000L) % 100L)
87
88 #define CHASTE_XSD_VERSION_MINOR \
89 ((XSD_INT_VERSION / 10000L) % 100L)
90
91 #define CHASTE_XSD_VERSION_PATCH \
92 ((XSD_INT_VERSION / 100L) % 100L)
93
94#else
95
96 EXCEPTION("Unsupported pre-release CodeSynthesis XSD version");
97
98#endif
99
100#define CHASTE_XSD_VERSION_AT_LEAST(MAJOR, MINOR, PATCH) \
101 ((CHASTE_XSD_VERSION_MAJOR > (MAJOR)) || \
102 (CHASTE_XSD_VERSION_MAJOR == (MAJOR) && \
103 CHASTE_XSD_VERSION_MINOR > (MINOR)) || \
104 (CHASTE_XSD_VERSION_MAJOR == (MAJOR) && \
105 CHASTE_XSD_VERSION_MINOR == (MINOR) && \
106 CHASTE_XSD_VERSION_PATCH >= (PATCH)))
107
108
109#endif /*CHASTEXSDVERSION_HPP_*/
